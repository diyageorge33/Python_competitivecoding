class Solution:
    def longestConsecutive(self, nums: List[int]) -> int:
        numset=set(nums)
        maxlen=0
        for n in numset:
            if n-1 not in numset:
                curnum=n
                curmaxlen=1
                while curnum+1 in numset:
                    curnum+=1
                    curmaxlen+=1
                maxlen=max(maxlen,curmaxlen)
        return maxlen
