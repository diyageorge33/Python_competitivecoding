class Solution:
    def twoSum(self, numbers: List[int], target: int) -> List[int]:
        first=0
        last=len(numbers)-1
        while first<last:
            x=numbers[first]+numbers[last]
            if x==target:
                return first+1,last+1
            elif x<target:
                first=first+1
            else:
                last=last-1
