class Solution:
    def moveZeroes(self, nums: List[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        right=0
        l=len(nums)
        for left in range(len(nums)):
            if nums[left]!=0:
                nums[right]=nums[left]
                right=right+1
        while right<l:
            nums[right]=0
            right=right+1
