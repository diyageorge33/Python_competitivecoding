BRUTE FORCE
class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        l=len(nums)
        for i in range(l-1):
            for j in range(i+1,l):
                if nums[i] + nums[j]==target:
                    return [i,j]
        return

class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        l=len(nums)
        seen={}
        for i in range(l):
            x=nums[i]
            complement=target-x
            if complement in seen:
                return [seen[complement],i]
            seen[x]=i


class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        dict1={}
        for i in range(len(nums)):
            diff=target-nums[i]
            if diff in dict1:
                return [dict1[diff],i]
            dict1[nums[i]]=i
