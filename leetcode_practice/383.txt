class Solution:
    def canConstruct(self, ransomNote: str, magazine: str) -> bool:
        dict1={}
        for i in magazine:
            if i not in dict1:
                dict1[i]=1
            else:
                dict1[i]=dict1[i]+1

        for j in ransomNote:
            if j in dict1:
                if dict1[j]>0:
                    dict1[j]=dict1[j]-1
                else:
                    return False
            else:
                return False
        return True
