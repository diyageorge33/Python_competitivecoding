class Solution:
    def waysToSplitArray(self, nums: list[int]) -> int:
        total_sum=sum(nums)
        split=0
        left_sum=0
        for i in range(0,len(nums)-1):
            left_sum=left_sum+nums[i]
            right_sum=total_sum-left_sum
            if left_sum>=right_sum:
                split=split+1

        return split
