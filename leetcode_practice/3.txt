class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        l=len(s)
        left=0
        d={}
        maxlen=0

        for right in range(0,l):
            if s[right] in d and d[s[right]]>=left:
                left=d[s[right]]+1

            d[s[right]]=right

            x=right-left+1
            maxlen=max(maxlen,x)
        return maxlen
