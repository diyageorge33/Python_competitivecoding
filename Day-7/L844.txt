class Solution:
    def backspaceCompare(self, s: str, t: str) -> bool:
        # st=[]
        def back(a):
            st=[]
            for i in a:
                if i!='#':
                    st.append(i)
                elif st:
                    st.pop()
            return st

        x=back(s)
        y=back(t)
        if x==y:
            return True
        else:
            return False
