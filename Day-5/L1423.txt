class Solution:
    def maxScore(self, cardPoints: List[int], k: int) -> int:
        sum1=sum(cardPoints[:k])
        i=k-1
        j=len(cardPoints)-1
        maxi=sum1
        while i>=0:
            sum1=sum1-cardPoints[i]
            sum1=sum1+cardPoints[j]
            if sum1>maxi:
                maxi=sum1
            i=i-1
            j=j-1
        return(maxi)
