class Solution:
    def mySqrt(self, x: int) -> int:
        start=1
        end=x
        if x==0:
            return 0
        while start<=end:
            m=(start+end)//2
            if (m*m)>x:
                end=m-1
            elif (m*m)<x:
                start=m+1
            else:
                return m
        return end
