class Solution:
    def numSubarrayProductLessThanK(self, nums: List[int], k: int) -> int:
        r,l=0,0
        count=0
        p=1
        if k<1:
            return 0
        while  r<len(nums):
            p=p*nums[r]
            while p>=k and l<=r:
                p=p//nums[l]
                l=l+1
            count=count+r-l+1
            r=r+1
        return count
